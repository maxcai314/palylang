// pretty assembly section
section .prettyasm

// returns ascii of number in a1
procedure binary_to_ascii:
  addi a1, a1, 48         
  jalr zero, ra           // return
endprocedure; 

procedure print_four_digit:
  // four digit integer to print stored in a0
  addi sp, sp, -16
  sw 4(sp), ra
  sw 0(sp), a0
  addi a2, zero, 10000
  rem a0, a0, a2
  addi a3, zero, 10
  div a2, a2, a3
@start:
  beq a2, zero, @end
  div a1, a0, a2
  jal ra, binary_to_ascii
  printc a1
  rem a0, a0, a2
  div a2, a2, a3
  jal zero, @start
@end:
  lw a0, 0(sp)
  lw ra, 4(sp)
  addi sp, sp, 16
  jalr zero, ra // return
endprocedure;

procedure print_str:      // prints the C-string at *a0
  addi    a2, a0, 0       // preserve a0
@start:
  lbu     a1, 0(a0)       // read byte
  beq     a1, zero, @end  // break if zero (null termination)
  printc  a1              // print char
  addi    a0, a0, 1       // increment
  jal     zero, @start    // jump to loop start
@end:
  addi    a0, a2, 0       // restore a0
  jalr    zero, ra        // return
endprocedure;

procedure init_array:
  la a0, r_arr            // a0 is pointer
  addi a1, a0, 11200      // a1 is end
  addi a2, zero, 2000     // val is 2000
@start:
  beq a0, a1, @end        // if ptr is 
  sw 0(a0), a2
  addi a0, a0, 4
  jal zero, @start
@end:
  sw 0(a0), zero
  jalr zero, ra
endprocedure;

// i: 0(sp), k: 4(sp), b: 8(sp), c: 12(sp), d: 16(sp), ra: 20(sp)
// meow meow uwu nya nyaa mrrrow :3
procedure print_pi:
  addi sp, sp, -32          // give stack space 
  sw 20(sp), ra
  sw 12(sp), zero
  addi a0, zero, 2800
  sw 4(sp), a0
  jal ra, init_array
@start_outer:
  lw a0, 4(sp)
  beq a0, zero, @end_outer  // k in a0

  addi a1, zero, 0          // d in a1 = 0
  addi a2, a0, 0            // i in a2 = k

  sw 16(sp), a1
  sw 0(sp), a2
@start_inner:
  la a3, r_arr              // load r_arr in a3
  lw a2, 0(sp)              // load i in a2
  add a3, a3, a2
  add a3, a3, a2
  add a3, a3, a2
  add a3, a3, a2            // get to r[i] pointer in a3

  lw a1, 0(a3)              // load r[i] in a1
  addi a0, zero, 10000
  mul a1, a0, a1            // rhs in a1
  lw a0, 16(sp)
  add a0, a0, a1            // d stored in a0
  // b = 2 * i - 1
  add a1, zero, a2
  add a1, a1, a2
  addi a1, a1, -1           // b stored in a1

  // r[i] = d % b;
  sw 0(sp), a2              // spill i back to stack; a2 is free
  rem a2, a0, a1
  sw 0(a3), a2
  // d /= b;
  div a0, a0, a1
  // i--;
  lw a2, 0(sp)
  addi a2, a2, -1

  sw 16(sp), a0
  sw 0(sp), a2
  sw 8(sp), a1              // store stuff in case we break

  beq a2, zero, @end_inner  // if (i == 0) break
  // d *= i;
  mul a0, a0, a2
  // store stuff: d, i, b
  sw 16(sp), a0
  sw 0(sp), a2
  sw 8(sp), a1
  
  jal zero, @start_inner
@end_inner:
  // printf("%.4d", c + d / 10000);
  lw a0, 12(sp) // c
  lw a1, 16(sp) // d
  addi a2, zero, 10000
  div a1, a1, a2
  add a0, a0, a1
  jal ra, print_four_digit

	// c = d % 10000;
  lw a1, 16(sp) // d
  addi a2, zero, 10000
  rem a0, a1, a2
  sw 12(sp), a0

  lw a0, 4(sp)
  addi a0, a0, -14          // subtract 14
  sw 4(sp), a0
  jal zero, @start_outer    // go back to outer loop
@end_outer:
  lw ra, 20(sp)             // load the return address
  addi sp, sp, 32           // restore stack
  jalr zero, ra             // return
endprocedure;
  


// raw assembly section
section .text

main:
  addi sp, sp, -16 // push 16 bytes of stack space
  sw 8(sp), ra // store the return address
  la a0, my_string
  jal ra, print_str // function call

  // jal ra, init_array
  jal ra, print_pi

  lw ra, 8(sp) // load the return address
  addi sp, sp, 16   // restore stack
  jalr zero, ra     // return


// this is a data section
section .data

// declare an array of zeros
r_arr:
  .zero 11204

// declare a C-string
my_string:
  .string "we//sh\n" // lol
