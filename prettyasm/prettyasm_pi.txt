// pretty assembly section
section .prettyasm

// returns ascii of number in a1
procedure binary_to_ascii:
  addi a1, a1, 48         
  jalr zero, ra           // return
endprocedure; 

procedure print_four_digit:
  // four digit integer to print stored in a0
  addi sp, sp, -16
  sw 4(sp), ra
  sw 0(sp), a0
  addi a2, zero, 10000
  rem a0, a0, a2
  addi a3, zero, 10
  div a2, a2, a3
@start:
  beq a2, zero, @end
  div a1, a0, a2
  jal ra, binary_to_ascii
  printc a1
  rem a0, a0, a2
  div a2, a2, a3
  jal zero, @start
@end:
  lw a0, 0(sp)
  lw ra, 4(sp)
  addi sp, sp, 16
  jalr zero, ra // return
endprocedure;

procedure print_str:      // prints the C-string at *a0
  addi    a2, a0, 0       // preserve a0
@start:
  lbu     a1, 0(a0)       // read byte
  beq     a1, zero, @end  // break if zero (null termination)
  printc  a1              // print char
  addi    a0, a0, 1       // increment
  jal     zero, @start    // jump to loop start
@end:
  addi    a0, a2, 0       // restore a0
  jalr    zero, ra        // return
endprocedure;

procedure init_array:
  la a0, r_arr            // a0 is pointer
  addi a1, a0, 11200      // a1 is end
  addi a2, zero, 2000     // val is 2000
@start:
  beq a0, a1, @end        // if ptr is 
  sw 0(a0), a2
  addi a0, a0, 4
  jal zero, @start
@end:
  sw 0(a0), zero
  jalr zero, ra
endprocedure;

// raw assembly section
section .text

main:
  addi sp, sp, -16 // push 16 bytes of stack space
  sw 8(sp), ra // store the return address
  la a0, my_string
  jal ra, print_str // function call

  addi a0, zero, 12345
  jal ra, print_four_digit

  jal ra, init_array
  la a1, r_arr
  lw a0, 0(a1)
  jal ra, print_four_digit

  lw ra, 8(sp) // load the return address
  addi sp, sp, 16   // restore stack
  jalr zero, ra     // return


// this is a data section
section .data

// declare an array of zeros
r_arr:
  .zero 11204

// declare a C-string
my_string:
  .string "we//sh\n" // lol
