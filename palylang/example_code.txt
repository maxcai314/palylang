// palylang code
// java-like, except all types are passed by value
// a type can be prepended with @ to indicate it is a pointer

// all struct/union types must be declared globally
// references are taken using @, and dereferenced using $
// since these operators don't overlap with any other operators,
// it should be easy to parse using ANTLR.

// builtin types int, long, short, void, char (1 byte)
// arrays can be declared locally, but are not actual types
// they're treated implicitly as an extra layer of indirection

void println(@char line) {
    while ($line != 0) {
        putchar(line);
        line++;
    }
    putchar('\n');
}
