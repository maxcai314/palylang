
// this code uses only integer arithmetic, so we can focus on a subset of the language for now
// booleans are also integer values, where zero is false and nonzero is true, just like in C

int sqrt(int n) {
    int INITIAL_GUESS = 1;
    int NUM_ITERATIONS = 10;

    int estimate = INITIAL_GUESS;
    // newton's method
    for (int i=0; i<NUM_ITERATIONS; i++) {
        estimate = (estimate + n / estimate) / 2;
    }
    return estimate;
}

int funny(int a, int b) {
    a += b;
    b = a - b;
    a -= b;
    return a;
}

int sillySquare(int n) {
    int total = 0;
    for (int i=0; i<n; i++) {
        total += n;
    }
    return total;
}

int waitWhy(int n) {

    }
}
